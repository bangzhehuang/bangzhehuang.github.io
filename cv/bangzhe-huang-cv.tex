\cvsection{Research Experience}

\project
  {AI4AI: Learning to Design a Model}
  {Lead researcher; advisor: Prof. Ziming Liu, College of AI, Tsinghua University; active pilot}

\begin{itemize}
  \item Built a controlled AI4AI pilot in which a graph meta-model learns the
  mapping from neural architectures to downstream test log-MAE under fixed
  regression tasks, data splits, optimizers, and training budgets.
  \item Defined a DAG architecture DSL from four homogeneous MLP blocks and one
  softmax gate, exhaustively enumerated 10,006 canonical candidates, and encoded
  component and edge structure with a graph neural network.
  \item Developed an interactive demo for manual DAG construction, DSL
  validation, outcome prediction, automatic architecture design, and training
  dynamics. Preliminary experiments identified a high-performing ResNet-like
  design.
\end{itemize}

\project
  {Scaling Foundation Models for Many-Body Physics}
  {Project lead; advisor: Prof. Jing Wang, Department of Physics, Fudan University; preprint in preparation}

\begin{itemize}
  \item Identified severe optimization ill-conditioning as a key barrier to
  scaling a monolithic model across heterogeneous Hamiltonian families, and
  proposed a task-modular architecture that preserves family-specific physical
  priors and ansatzes while enabling cross-task knowledge composition.
  \item Led problem formulation, method design, PyTorch implementation, and
  evaluation against unified joint-training baselines across task scales, task
  combinations, and fine-tuning settings.
  \item Preliminary results show comparable generalization and adaptation when
  unified training remains tractable, while the modular system remains
  trainable in regimes where the unified baseline becomes ill-conditioned.
\end{itemize}

\project
  {Open-Ended Scientific Research Benchmark for Physics Research Agents}
  {Independent research; academic discussion: Prof. Siheng Chen, School of Artificial Intelligence, Shanghai Jiao Tong University; pilot completed}

\begin{itemize}
  \item Designed open-ended physics research tasks with stable ground truth
  defined by low-dimensional physical structures and equivalent observables,
  without prescribing a single valid research trajectory.
  \item Built an end-to-end pilot spanning experiment and simulation design,
  code generation and execution, multi-round observation and feedback, and a
  final physical audit.
\end{itemize}

\project
  {Phase Transport and Defect Dynamics in Nonreciprocal Spin Systems}
  {Lead researcher; advisor: Prof. Zi Cai, School of Physics and Astronomy, Shanghai Jiao Tong University; pilot completed}

\begin{itemize}
  \item Used Monte Carlo simulations to study the microscopic origin of
  macroscopic oscillations in two-dimensional nonreciprocal spin systems and
  extracted oscillation frequency, domain-wall density, and defect observables.
  \item Connected oscillation frequency $\omega$, domain-wall density $\rho$,
  and effective transport velocity through $\omega=\rho v_{\mathrm{eff}}$, and
  extracted $v_{\mathrm{eff}}$ from the linear $\omega$--$\rho$ regime.
  \item Identified regimes dominated by vortex-pair generation and periodic
  wall expansion versus vortex-constrained topological structures, and tested
  the physical picture across lattice geometries and symmetric bond disorder.
\end{itemize}
